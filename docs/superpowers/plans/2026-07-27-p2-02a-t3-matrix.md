# P2-02A T3 — Real A–F GPU Matrix Runner Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a thin T3 runner that preflights the GPU host, runs the frozen T2 watchdog over the A–F matrix, classifies the result, and emits a machine-readable T4 (100-batch) authorization gate — without touching any T1/T2 frozen code.

**Architecture:** Approach A (thin wrapper). `matrix_cli` → `matrix_runner.run_matrix` runs `preflight` hard gates, captures an `environment` fingerprint, calls the frozen `watchdog_cli.main()` in-process (real probe path), reads `_watchdog_summary.json`, classifies per-cell `CellVerdict`s into a T4 gate, and atomically writes `t3_matrix_report.json`. GPU/VRAM/weights probing goes through an injectable adapter so everything except the real GPU run is unit/integration-testable on a CPU-only host.

**Tech Stack:** Python 3.11, stdlib (`argparse`, `json`, `dataclasses`, `pathlib`, `os`, `socket`, `platform`, `tempfile`), pytest. Real adapter uses `torch`/`torchvision` lazily (GPU host only). Reuses frozen `p1.stability.watchdog_cli`, `p1.stability.schema.CellVerdict`, `p1.stability.protocol.load_p202a_protocol`, `p1.training.frozen_inputs`.

## Global Constraints

- **No edits to frozen code:** `p1/stability/{cli,probe,stages,schema,matrix,protocol,watchdog,watchdog_cli}.py`, `p1/{audit,manifest,split}.py`, `p1/benchmark/**`, `protocol_p2_02a_v1.json`, frozen split/fold CSVs. T3 is **additive only**.
- **No T4 run:** never execute the 100-batch loop.
- **No CPU-only / low-VRAM remediation:** fail closed. No CPU fallback, no auto batch-size shrink, no backbone downgrade.
- **No final `Decision` enum / `decision.json`:** do NOT call `schema.finalize_decision`; T3 is matrix-only. `t4_gate` is a T3-local string, never a `schema.Decision` member.
- **Serial strategy = full diagnostic map:** run all A–F (delegated to frozen `watchdog_cli`); never short-circuit.
- **T4 gate predicate:** `T4_AUTHORIZED` iff `matrix_verdicts["E"] == "PASS"` AND `matrix_verdicts["F"] == "PASS"`, else `T4_BLOCKED`.
- **Do not commit** `algorithm/.omc/**`, `algorithm/artifacts/**`, `algorithm/datasets/**`, real-run JSONs, console tee logs.
- **New modules under** `algorithm/p1/stability/`; **new tests under** `algorithm/eye_diagnosis_p1/tests/`.
- **CI:** T3 unit tests run in the normal `python` job (`-m "not slow"`); T3 integration test is `slow` and runs in the dedicated `watchdog` job. No GPU in CI.
- **Test command (from `algorithm/`):** `python -m pytest eye_diagnosis_p1/tests/test_p2_02a_t3_*.py -q`.

# environment.py
from __future__ import annotations
import platform
import socket
import sys
from typing import Optional, Protocol


class T3EnvAdapter(Protocol):
    def cuda_is_available(self) -> bool: ...
    def cuda_device_count(self) -> int: ...
    def cuda_mem_get_info(self, device: int = 0) -> tuple[int, int]: ...
    def device_name(self, device: int = 0) -> str: ...
    def versions_dict(self) -> dict: ...
    def weights_cache_path(self, weights_id: str) -> Optional[str]: ...


def capture_environment(adapter: T3EnvAdapter, *, now: str, git_commit: Optional[str]) -> dict:
    """Read-only environment fingerprint. Never raises for missing GPU."""
    available = bool(adapter.cuda_is_available())
    free_mb = total_mb = 0
    device_name = ""
    if available and adapter.cuda_device_count() >= 1:
        free_b, total_b = adapter.cuda_mem_get_info(0)
        free_mb, total_mb = free_b // (1024**2), total_b // (1024**2)
        device_name = adapter.device_name(0)
    v = adapter.versions_dict()
    return {
        "timestamp": now,
        "git_commit": git_commit,
        "cuda_available": available,
        "device_count": int(adapter.cuda_device_count()),
        "device_name": device_name,
        "vram_free_mb": int(free_mb),
        "vram_total_mb": int(total_mb),
        "torch_version": v.get("torch", ""),
        "cuda_version": v.get("cuda", ""),
        "cudnn_version": v.get("cudnn", ""),
        "driver_version": v.get("driver", ""),
        "python_version": sys.version.split()[0],
        "platform": platform.platform(),
        "hostname": socket.gethostname(),
    }


class RealT3EnvAdapter:
    """torch/torchvision-backed adapter. Imported lazily on the GPU host."""

    def cuda_is_available(self) -> bool:
        import torch
        return bool(torch.cuda.is_available())

    def cuda_device_count(self) -> int:
        import torch
        return int(torch.cuda.device_count())

    def cuda_mem_get_info(self, device: int = 0) -> tuple[int, int]:
        import torch
        return tuple(torch.cuda.mem_get_info(device))  # type: ignore[return-value]

    def device_name(self, device: int = 0) -> str:
        import torch
        return str(torch.cuda.get_device_name(device))

    def versions_dict(self) -> dict:
        import torch
        return {
            "torch": torch.__version__,
            "cuda": torch.version.cuda or "",
            "cudnn": str(getattr(torch.backends.cudnn, "version", lambda: "")() or ""),
            "driver": _driver_version(),
        }

    def weights_cache_path(self, weights_id: str) -> Optional[str]:
        # weights_id e.g. "EfficientNet_B4_Weights.IMAGENET1K_V1"
        import os
        import torch
        from torchvision.models import get_weight
        try:
            weight = get_weight(weights_id)
            filename = weight.url.split("/")[-1]
        except Exception:
            return None
        candidate = os.path.join(torch.hub.get_dir(), "checkpoints", filename)
        return candidate if os.path.exists(candidate) else None


def _driver_version() -> str:
    try:
        import torch
        v = torch._C._cuda_getDriverVersion()  # type: ignore[attr-defined]
        return str(v)
    except Exception:
        return ""
```

# preflight.py
from __future__ import annotations
import shutil
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional

from p1.stability.environment import T3EnvAdapter


@dataclass(frozen=True)
class CheckResult:
    name: str
    ok: bool
    detail: str


@dataclass(frozen=True)
class PreflightResult:
    ok: bool
    checks: list[CheckResult]
    reason: Optional[str]


def run_preflight(
    *,
    protocol_path: Path,
    manifest_path: Path,
    split_path: Path,
    fold_path: Path,
    image_root: Path,
    out_dir: Path,
    vram_floor_mb: int,
    weights_id: str,
    adapter: T3EnvAdapter,
) -> PreflightResult:
    checks: list[CheckResult] = []

    # 1. GPU availability (hard gate; no CPU fallback).
    gpu_ok = bool(adapter.cuda_is_available()) and adapter.cuda_device_count() >= 1
    checks.append(CheckResult("gpu", gpu_ok,
                              f"cuda_available={adapter.cuda_is_available()} devices={adapter.cuda_device_count()}"))

    # 2. VRAM floor (hard gate; no auto batch shrink).
    vram_ok = False
    if gpu_ok:
        free_b, total_b = adapter.cuda_mem_get_info(0)
        free_mb = free_b // (1024**2)
        vram_ok = free_mb >= vram_floor_mb
        checks.append(CheckResult("vram", vram_ok, f"free_mb={free_mb} floor_mb={vram_floor_mb}"))
    else:
        checks.append(CheckResult("vram", False, "skipped: no gpu"))

    # 3. Frozen-input integrity (reuse P2-01 validators; no modification).
    fi_ok, fi_detail = _check_frozen_inputs(protocol_path, manifest_path, split_path, fold_path)
    checks.append(CheckResult("frozen_inputs", fi_ok, fi_detail))

    # 4. Weights cached + image-root readable + out-dir writable + disk free.
    wp_ok, wp_detail = _check_weights_paths_disk(adapter, weights_id, image_root, out_dir)
    checks.append(CheckResult("weights_paths_disk", wp_ok, wp_detail))

    reason = None
    if not gpu_ok:
        reason = "no_cuda"
    elif not vram_ok:
        reason = "insufficient_vram"
    elif not fi_ok:
        reason = "frozen_input_integrity"
    elif not wp_ok:
        reason = "weights_paths_disk"
    return PreflightResult(ok=(reason is None), checks=checks, reason=reason)


def _check_frozen_inputs(protocol_path, manifest_path, split_path, fold_path) -> tuple[bool, str]:
    try:
        from p1.stability.protocol import load_p202a_protocol
        from p1.training.frozen_inputs import (
            load_split_assignment, load_fold_assignment, validate_frozen_training_inputs,
        )
        import pandas as pd
        load_p202a_protocol(Path(protocol_path))
        manifest_pids = set(pd.read_parquet(manifest_path)["patient_id"].astype(str).tolist())
        split = load_split_assignment(Path(split_path))
        fold = load_fold_assignment(Path(fold_path), expected_folds=5)
        validate_frozen_training_inputs(manifest_pids, split, fold, expected_folds=5)
        return True, "protocol+manifest+split+fold validated"
    except Exception as e:  # P202AProtocolError / FrozenInputError / IO
        return False, f"{type(e).__name__}: {e}"


def _check_weights_paths_disk(adapter, weights_id, image_root, out_dir) -> tuple[bool, str]:
    image_root, out_dir = Path(image_root), Path(out_dir)
    if not image_root.is_dir():
        return False, f"image_root not readable: {image_root}"
    try:
        out_dir.mkdir(parents=True, exist_ok=True)
        probe = out_dir / ".t3_write_probe"
        probe.write_text("x", encoding="utf-8"); probe.unlink()
    except OSError as e:
        return False, f"out_dir not writable: {e}"
    free_bytes = shutil.disk_usage(out_dir).free
    if free_bytes < 1 * 1024**3:  # 1 GiB floor for logs/reports
        return False, f"insufficient disk: {free_bytes // 1024**2} MB free"
    wp = adapter.weights_cache_path(weights_id)
    if not wp:
        return False, f"weights not cached for {weights_id}"
    return True, f"weights={wp}"

T4_AUTHORIZED = "T4_AUTHORIZED"
T4_BLOCKED = "T4_BLOCKED"
MISSING = "MISSING"
_PASS = CellVerdict.PASS.value


@dataclass(frozen=True)
class ClassifyResult:
    matrix_verdicts: dict[str, str]
    t4_gate: str
    blocking_cells: list[str]
    diagnostic: str


def classify(summary: dict) -> ClassifyResult:
    labels = [c.label for c in expand_matrix()]  # canonical A..F
    verdicts = {lab: str(summary.get(lab, {}).get("verdict", MISSING)) for lab in labels}
    blocking = [lab for lab in ("E", "F") if verdicts.get(lab) != _PASS]
    gate = T4_AUTHORIZED if not blocking else T4_BLOCKED
    return ClassifyResult(
        matrix_verdicts=verdicts,
        t4_gate=gate,
        blocking_cells=blocking,
        diagnostic=_diagnostic(verdicts, blocking),
    )


def _diagnostic(verdicts: dict[str, str], blocking: list[str]) -> str:
    # Borrows the intent of schema.finalize_decision WITHOUT calling it
    # (batch100 is absent at T3; calling it would mislabel all-PASS as BLOCKED).
    if verdicts.get("A") == CellVerdict.CUDA_ERROR.value:
        return "CUDA runtime blocked (cell A failed the no-AMP/batch-1 baseline)"
    oom = [lab for lab, v in verdicts.items() if v == CellVerdict.OOM.value]
    if oom:
        return f"memory-bound (OOM at cell {oom[0]})"
    if verdicts.get("E") == _PASS and verdicts.get("F") != _PASS:
        return "target DataLoader config (workers/pin) unstable at cell F"
    if not blocking:
        return "matrix clean; T4 authorized for E and F"
    return f"T4 blocked; non-PASS target cells: {blocking}"


def _write_json_atomic(path: Path, payload: dict) -> None:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=str(path.parent), suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as fh:
            fh.write(json.dumps(payload, indent=2, sort_keys=True) + "\n")
        os.replace(tmp, path)
    finally:
        if os.path.exists(tmp):
            os.remove(tmp)
```

from dataclasses import dataclass as _dataclass, field as _field

from p1.stability.environment import T3EnvAdapter, capture_environment
from p1.stability.preflight import run_preflight, PreflightResult
from p1.stability import watchdog_cli

T3_REPORT_SCHEMA = "p2-02a-t3-v1"


@_dataclass
class MatrixConfig:
    protocol_path: Path
    manifest_path: Path
    split_path: Path
    fold_path: Path
    image_root: Path
    out_dir: Path
    vram_floor_mb: int
    weights_id: str
    heartbeat_timeout: float
    process_timeout: float
    now: str
    git_commit: _Optional[str] = None
    child: _Optional[Path] = None


@_dataclass
class MatrixReport:
    schema_version: str
    ok: bool
    exit_code: int
    t4_gate: str
    report_path: Path
    preflight: PreflightResult
    environment: dict


def _preflight_dict(pf: PreflightResult) -> dict:
    return {"ok": pf.ok, "reason": pf.reason,
            "checks": [{"name": c.name, "ok": c.ok, "detail": c.detail} for c in pf.checks]}


def run_matrix(cfg: MatrixConfig, adapter: T3EnvAdapter) -> MatrixReport:
    cfg.out_dir.mkdir(parents=True, exist_ok=True)
    env = capture_environment(adapter, now=cfg.now, git_commit=cfg.git_commit)
    _write_json_atomic(cfg.out_dir / "environment.json", env)

    pf = run_preflight(
        protocol_path=cfg.protocol_path, manifest_path=cfg.manifest_path,
        split_path=cfg.split_path, fold_path=cfg.fold_path, image_root=cfg.image_root,
        out_dir=cfg.out_dir, vram_floor_mb=cfg.vram_floor_mb, weights_id=cfg.weights_id,
        adapter=adapter,
    )
    _write_json_atomic(cfg.out_dir / "preflight.json", _preflight_dict(pf))

    if not pf.ok:
        report_path = cfg.out_dir / "t3_matrix_report.json"
        _write_report(report_path, env, pf, matrix_verdicts={}, t4_gate=T4_BLOCKED,
                      blocking_cells=[], diagnostic=f"preflight failed: {pf.reason}")
        return MatrixReport(T3_REPORT_SCHEMA, False, 2, T4_BLOCKED, report_path, pf, env)

    # Delegate the full A-F serial loop to the FROZEN T2 watchdog (in-process).
    argv = [
        "--protocol", str(cfg.protocol_path), "--manifest", str(cfg.manifest_path),
        "--split-assignment", str(cfg.split_path), "--fold-assignment", str(cfg.fold_path),
        "--image-root", str(cfg.image_root), "--out-dir", str(cfg.out_dir),
        "--heartbeat-timeout", str(cfg.heartbeat_timeout),
        "--process-timeout", str(cfg.process_timeout),
    ]
    if cfg.child is not None:
        argv += ["--child", str(cfg.child)]
    watchdog_cli.main(argv)

    summary = json.loads((cfg.out_dir / "_watchdog_summary.json").read_text(encoding="utf-8"))
    cr = classify(summary)
    report_path = cfg.out_dir / "t3_matrix_report.json"
    _write_report(report_path, env, pf, matrix_verdicts=cr.matrix_verdicts, t4_gate=cr.t4_gate,
                  blocking_cells=cr.blocking_cells, diagnostic=cr.diagnostic)
    exit_code = 0 if cr.t4_gate == T4_AUTHORIZED else 1
    return MatrixReport(T3_REPORT_SCHEMA, cr.t4_gate == T4_AUTHORIZED, exit_code, cr.t4_gate,
                        report_path, pf, env)

    _write_json_atomic(path, {
        "schema_version": T3_REPORT_SCHEMA,
        "timestamp": env.get("timestamp"),
        "environment": env,
        "preflight": _preflight_dict(pf),
        "matrix_verdicts": matrix_verdicts,
        "t4_gate": t4_gate,
        "blocking_cells": blocking_cells,
        "diagnostic": diagnostic,
        "watchdog_summary_ref": "_watchdog_summary.json",
    })

# matrix_cli.py
from __future__ import annotations
import argparse
import datetime as _dt
import subprocess
import sys
from pathlib import Path
from typing import Optional

from p1.stability.matrix_runner import MatrixConfig, run_matrix


def _build_argparser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(
        prog="p1.stability.matrix_cli",
        description="T3: preflight + frozen-watchdog A-F matrix + T4 gate. Writes t3_matrix_report.json.",
    )
    p.add_argument("--protocol", type=Path, required=True)
    p.add_argument("--manifest", type=Path, required=True)
    p.add_argument("--split-assignment", type=Path, required=True)
    p.add_argument("--fold-assignment", type=Path, required=True)
    p.add_argument("--image-root", type=Path, required=True)
    p.add_argument("--out-dir", type=Path, required=True)
    p.add_argument("--vram-floor-mb", type=int, default=6000)
    p.add_argument("--weights-id", type=str, default="EfficientNet_B4_Weights.IMAGENET1K_V1")
    p.add_argument("--heartbeat-timeout", type=float, default=90.0)
    p.add_argument("--process-timeout", type=float, default=300.0)
    p.add_argument("--now", type=str, default=None, help="ISO8601 timestamp; default = now (UTC).")
    p.add_argument("--child", type=Path, default=None, help="INTERNAL: fake harness (test only).")
    return p


def _git_commit() -> Optional[str]:
    try:
        out = subprocess.run(["git", "rev-parse", "HEAD"], check=False,
                             capture_output=True, text=True)
        return out.stdout.strip() or None
    except OSError:
        return None


def _config_from_args(args: argparse.Namespace) -> MatrixConfig:
    now = args.now or _dt.datetime.now(_dt.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
    return MatrixConfig(
        protocol_path=args.protocol, manifest_path=args.manifest,
        split_path=args.split_assignment, fold_path=args.fold_assignment,
        image_root=args.image_root, out_dir=args.out_dir,
        vram_floor_mb=args.vram_floor_mb, weights_id=args.weights_id,
        heartbeat_timeout=args.heartbeat_timeout, process_timeout=args.process_timeout,
        now=now, git_commit=_git_commit(), child=args.child,
    )


def main(argv: Optional[list[str]] = None) -> int:
    args = _build_argparser().parse_args(argv)
    from p1.stability.environment import RealT3EnvAdapter
    cfg = _config_from_args(args)
    report = run_matrix(cfg, RealT3EnvAdapter())
    return report.exit_code


if __name__ == "__main__":  # pragma: no cover
    sys.exit(main())


__all__ = ["main"]
